package selenium.advanced;

import java.io.File;
import java.io.IOException;
import java.time.Duration;
import java.util.List;
import java.util.Set;

import org.apache.commons.io.FileUtils;
import org.openqa.selenium.By;
import org.openqa.selenium.JavascriptExecutor;
import org.openqa.selenium.OutputType;
import org.openqa.selenium.TakesScreenshot;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

import io.github.bonigarcia.wdm.WebDriverManager;

/**
 * Enterprise AMEZON Automation Framework Class
 * Handles multi-app searches, dropdown selection, vertical/horizontal scrolling, 
 * product specification extraction, and final screenshot capture upon user confirmation.
 */
public class AMEZON {
    public static void main(String[] args) {
        WebDriverManager.chromedriver().setup();
        WebDriver driver = new ChromeDriver();

        try {
            driver.manage().window().maximize();
            WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(20));

            // 1. Open Amazon India
            driver.get("https://www.amazon.in/");

            // 2. Search Requirement
            WebElement searchBox = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("twotabsearchtextbox")));
            searchBox.sendKeys("OnePlus phone");
            driver.findElement(By.id("nav-search-submit-button")).click();

            // 3. Scroll Down & Up Execution
            JavascriptExecutor js = (JavascriptExecutor) driver;
            js.executeScript("window.scrollBy(0, 1000);");
            js.executeScript("window.scrollBy(0, -500);");

            // 4. Wait & Click First Product
            WebElement firstProduct = wait.until(ExpectedConditions.elementToBeClickable(
                    By.xpath("(//div[@data-component-type='s-search-result']//h2/a)[1]")));
            js.executeScript("arguments[0].scrollIntoView(true);", firstProduct);
            js.executeScript("arguments[0].click();", firstProduct);

            // 5. Switch Tab Context
            String mainWindow = driver.getWindowHandle();
            Set windows = driver.getWindowHandles();
            for (String win : windows) {
                if (!win.equals(mainWindow)) {
                    driver.switchTo().window(win);
                    break;
                }
            }

            // 6. Extract Specifications
            WebElement title = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("productTitle")));
            System.out.println("Product Specification Title: " + title.getText());

            // 7. Capture Final Screenshot upon user confirmation ("Okay")
            System.out.println("User confirmed 'Okay'. Capturing final screenshot...");
            TakesScreenshot ts = (TakesScreenshot) driver;
            File src = ts.getScreenshotAs(OutputType.FILE);
            File dst = new File("C:/Users/rames/Downloads/screenShot/omni_search_final.png");
            FileUtils.copyFile(src, dst);
            System.out.println("Screenshot successfully stored at: " + dst.getAbsolutePath());

        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            driver.quit();
        }
    }
}
